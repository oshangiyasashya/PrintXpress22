    @Insert(onConflict = OnConflictStrategy.ABORT)

    @Query("SELECT * FROM orders WHERE userId = :userId ORDER BY createdAt DESC")

    @Query("UPDATE orders SET status = :status WHERE id = :orderId")

package com.printxpress.data.db;

import android.content.Context;
import androidx.annotation.NonNull;
import androidx.room.Database;
import androidx.room.Room;
import androidx.room.RoomDatabase;
import androidx.sqlite.db.SupportSQLiteDatabase;
import com.printxpress.R;
import com.printxpress.data.entity.Address;
import com.printxpress.data.entity.Category;
import com.printxpress.data.entity.Faq;
import com.printxpress.data.entity.Notification;
import com.printxpress.data.entity.Order;
import com.printxpress.data.entity.OrderItem;
import com.printxpress.data.entity.OrderStatusHistory;
import com.printxpress.data.entity.Product;
import com.printxpress.data.entity.ProductOption;
import com.printxpress.data.entity.Promotion;
import com.printxpress.data.entity.SampleDesign;
import com.printxpress.data.entity.SavedDesign;
import com.printxpress.data.entity.User;
import java.util.concurrent.ExecutorService;
import java.util.concurrent.Executors;

@Database(
        entities = {
                User.class, Address.class, Category.class, Product.class,
                ProductOption.class, Order.class, OrderItem.class,
                OrderStatusHistory.class, SavedDesign.class, Notification.class,
                Promotion.class, Faq.class, SampleDesign.class
        },
        version = 1,
        exportSchema = false
)
public abstract class AppDatabase extends RoomDatabase {

    public abstract UserDao userDao();
    public abstract CatalogDao catalogDao();
    public abstract OrderDao orderDao();
    public abstract CommonDao commonDao();

    private static volatile AppDatabase INSTANCE;
    private static final int NUMBER_OF_THREADS = 4;
    public static final ExecutorService databaseWriteExecutor = Executors.newFixedThreadPool(NUMBER_OF_THREADS);

    public static AppDatabase getDatabase(final Context context) {
        if (INSTANCE == null) {
            synchronized (AppDatabase.class) {
                if (INSTANCE == null) {
                    INSTANCE = Room.databaseBuilder(
                                    context.getApplicationContext(),
                                    AppDatabase.class,
                                    "printxpress_database"
                            )
                            .addCallback(sRoomDatabaseCallback)
                            .build();
                }
            }
        }
        return INSTANCE;
    }

    private static final RoomDatabase.Callback sRoomDatabaseCallback = new RoomDatabase.Callback() {
        @Override
        public void onCreate(@NonNull SupportSQLiteDatabase db) {
            super.onCreate(db);
            databaseWriteExecutor.execute(() -> {
                if (INSTANCE == null) return;

                CatalogDao catalogDao = INSTANCE.catalogDao();
                CommonDao commonDao = INSTANCE.commonDao();

                // 1. Seed Categories
                Category catCards = new Category("Business Cards", android.R.drawable.ic_menu_agenda);
                Category catFlyers = new Category("Flyers & Brochures", android.R.drawable.ic_menu_gallery);
                Category catTshirts = new Category("Custom Apparel", android.R.drawable.ic_menu_share);
                Category catBanners = new Category("Banners & Signs", android.R.drawable.ic_menu_view);

                long[] catIds = catalogDao.insertCategories(catCards, catFlyers, catTshirts, catBanners);

                // 2. Seed Products
                Product p1 = new Product(catIds[0], "Standard Business Cards", "Premium cardstock with sharp print quality.", 1500.00, android.R.drawable.ic_menu_agenda, "350gsm Matte Artboard", "85x55mm");
                Product p2 = new Product(catIds[1], "A4 Promotional Flyers", "Full-color gloss flyers for mass distribution.", 3500.00, android.R.drawable.ic_menu_gallery, "150gsm Gloss Art Paper", "A4");
                Product p3 = new Product(catIds[2], "Custom Printed T-Shirt", "100% combed cotton with screen or DTG print.", 2200.00, android.R.drawable.ic_menu_share, "100% Cotton", "S, M, L, XL");

                long[] prodIds = catalogDao.insertProducts(p1, p2, p3);

                // 3. Seed Product Options
                catalogDao.insertOptions(
                        new ProductOption(prodIds[0], "Finish", "Matte Lamination", 300.00),
                        new ProductOption(prodIds[0], "Finish", "Gloss Lamination", 300.00),
                        new ProductOption(prodIds[0], "Corners", "Rounded Corners", 200.00),
                        new ProductOption(prodIds[1], "Paper Type", "150gsm Gloss", 0.00),
                        new ProductOption(prodIds[1], "Paper Type", "250gsm Premium", 800.00),
                        new ProductOption(prodIds[2], "Size", "Medium (M)", 0.00),
                        new ProductOption(prodIds[2], "Size", "Extra Large (XL)", 200.00)
                );

                // 4. Seed FAQs
                commonDao.insertFaqs(
                        new Faq("What is the standard turnaround time?", "Standard printing takes 2-3 business days. Rush orders can be completed in 24 hours."),
                        new Faq("What file formats do you accept for design uploads?", "We accept PDF, PNG, JPG, and AI formats with a minimum resolution of 300 DPI."),
                        new Faq("Can I collect my order in person?", "Yes, select 'Store Pickup' during checkout to pick up your order directly from our print studio.")
                );

                // 5. Seed Promotions
                long now = System.currentTimeMillis();
                long oneMonth = 30L * 24 * 60 * 60 * 1000;
                commonDao.insertPromotions(
                        new Promotion("First Order Special", "Get 15% off your very first order with PrintXpress!", 15.0, 1, now, now + oneMonth),
                        new Promotion("Bulk Order Discount", "Save 25% when ordering 500 units or more.", 25.0, 500, now, now + oneMonth)
                );

                // 6. Seed Sample Designs
                catalogDao.insertSampleDesigns(
                        new SampleDesign(catIds[0], "Minimalist Corporate Card", android.R.drawable.ic_menu_agenda),
                        new SampleDesign(catIds[0], "Bold Geometric Card", android.R.drawable.ic_menu_agenda),
                        new SampleDesign(catIds[1], "Event Showcase Flyer", android.R.drawable.ic_menu_gallery)
                );
            });
        }
    };
}
